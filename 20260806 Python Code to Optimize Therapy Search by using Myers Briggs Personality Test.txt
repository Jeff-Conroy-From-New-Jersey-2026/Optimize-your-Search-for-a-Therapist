def best_matches(patient_type, top_n=5):
    """Return the top therapist matches for a given patient MBTI type."""
    scores = compatibility_matrix.get(patient_type)
    if not scores:
        raise ValueError("Invalid MBTI type")

    # Sort therapist types by compatibility score (descending)
    sorted_matches = sorted(scores.items(), key=lambda x: x[1], reverse=True)

    return sorted_matches[:top_n]


# Example usage:
patient = "INFP"
print(f"Best therapist matches for {patient}:")
for therapist, score in best_matches(patient):
    print(f"{therapist}: {score}")
